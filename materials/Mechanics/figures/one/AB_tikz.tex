% Simple sketch: a curve y=f(x) joining two points A and B in the plane,
% with standard x/y axes.
%
% Compile standalone with:  pdflatex fx_curve.tex
\documentclass[tikz,border=6pt]{standalone}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, calc}

\begin{document}
\begin{tikzpicture}[>=Stealth, thick, blue, every node/.style={font=\fontsize{20}{24}\selectfont}]

  % ---- axes ---------------------------------------------------------
  \draw[->] (-0.6,0) -- (7.6,0) node[right] {$x$};
  \draw[->] (0,-0.6) -- (0,5.4) node[above] {$y$};

  % ---- endpoints ------------------------------------------------------
  \coordinate (A) at (2.3,1.15);
  \coordinate (B) at (5.6,4.3);

  % ---- curve y=f(x) from A to B, with a small hump/dip then a rise ------
  \draw (A)
    .. controls (2.6,1.90) and (2.9,2.05) .. (3.3,1.95)
    .. controls (3.7,1.85) and (4.0,1.70) .. (4.4,1.75)
    .. controls (4.9,1.85) and (5.25,3.05) .. (B);

  \fill (A) circle (2.2pt);
  \fill (B) circle (2.2pt);

  \node at ($(A)+(-0.35,-0.55)$) {$A$};
  \node at ($(B)+(0.45,0.35)$) {$B$};
  \node at (3.85,3.05) {$f(x)$};

\end{tikzpicture}
\end{document}
