\documentclass[tikz,border=6pt]{standalone}
\usepackage{amsmath}
\usepackage[UTF8,fontset=fandol]{ctex}
\usepackage{fix-cm}
\usetikzlibrary{arrows.meta}

\begin{document}
\begin{tikzpicture}[>=Stealth,thick,every node/.style={font=\fontsize{15}{18}\selectfont}]
  % Frequency response.
  \begin{scope}[xshift=-49pt, yshift=-37pt, xscale=1.34, yscale=1.34]
    \draw[blue!70!black,->] (-.25,0) -- (5.35,0) node[right] {$\gamma/\omega$};
    \draw[blue!70!black,->] (0,-.25) -- (0,4.75) node[above] {$|b|$};
    \draw[red,domain=.05:.92,samples=120,smooth]
      plot ({2.35*\x},{min(4.35,.72/abs(1-\x*\x))});
    \draw[red,domain=1.08:2.05,samples=140,smooth]
      plot ({2.35*\x},{min(4.35,.72/abs(1-\x*\x))});
    \draw[densely dashed] (2.35,0) -- (2.35,4.45);
    \node[below] at (2.35,0) {$1$};
    \node[red, align=center] at (4.46,3.46)
      {$|b|=\left|\dfrac{f}{m(\omega^2-\gamma^2)}\right|$};
    \node[green!40!black,align=center] at (1.18,4.35) {共振\\$\gamma=\omega$};
  \end{scope}

  % Resonant time-domain response.
  \begin{scope}[xshift=211pt, yshift=60pt, xscale=1.13, yscale=1.04]
    \draw[blue!70!black,->] (-.25,0) -- (6.15,0) node[right] {$t$};
    \draw[blue!70!black,->] (0,-3.28) -- (0,2.65) node[above] {$x$};
    \draw[red,domain=0:5.75,samples=300,smooth]
      plot (\x,{.38*\x*sin(250*\x)});
    \draw[green!45!black,densely dashed,domain=0:5.75,samples=80]
      plot (\x,{.38*\x});
    \draw[green!45!black,densely dashed,domain=0:5.75,samples=80]
      plot (\x,{-.38*\x});
    \node[green!40!black,fill=white,inner sep=2pt] at (3.52,2.28)
      {$\displaystyle \pm\frac{f}{2m\omega}t$};
    \node[red,align=center] at (3.0,-3.15) {共振时：外力在每个周期\\同相输入能量};
  \end{scope}
\end{tikzpicture}
\end{document}
